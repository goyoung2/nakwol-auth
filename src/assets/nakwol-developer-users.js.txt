import { NakwolAuthClient } from '/sdk/v0.3.2/nakwol-auth-web.js';
const auth=new NakwolAuthClient({clientId:'nakwol-connect-admin',redirectUri:location.origin+'/admin/apps',authOrigin:location.origin});
const $=id=>document.getElementById(id),labels={authorized:'인증 완료 · 방문 미관측',observed:'접근 관측',denied:'인증 거부',blocked:'앱 차단',preregistered:'사전등록'};
let selected='',user=null,cursor=null,generation=0,busy=false,pending=null,capabilities=null;
const base=()=>'/developer/v1/apps/'+encodeURIComponent(selected);
function status(text){$('status').textContent=text;}
function date(value){return value?new Date(value).toLocaleString():'미확인';}
function el(tag,text){const node=document.createElement(tag);node.textContent=text;return node;}
async function api(path,body){const token=auth.getAccessToken();if(!token)throw Object.assign(new Error('관리자 인증이 필요합니다.'),{code:'UNAUTHENTICATED'});const response=await fetch(path,{method:body?'POST':'GET',cache:'no-store',headers:{Authorization:'Bearer '+token,'Content-Type':'application/json'},body:body?JSON.stringify(body):undefined});const json=await response.json();if(!response.ok)throw Object.assign(new Error(json.error?.code||'요청 실패'),{code:json.error?.code,status:response.status});return json.data;}
function fail(error){if(error.status===401||/AUTH/.test(error.code||'')){$('login').hidden=false;status('관리자 인증을 갱신한 뒤 다시 시도하세요.');}else if(error.status===409){pending=null;status('상태나 관리 권한이 변경되었습니다. 새로고침 후 다시 확인하세요.');}else status(error.message);}
function editable(){for(const id of ['service','filter','reload','more','action-fields'])$(id).disabled=busy;}
async function show(subject){const current=++generation;$('detail').hidden=true;pending=null;user=null;try{const value=await api(base()+'/users/'+encodeURIComponent(subject));if(current!==generation)return;user=value;$('user-title').textContent=(value.displayName||'사전등록 사용자')+' · '+value.discordId;$('user-detail').textContent=labels[value.state]+' · 최초 승인 '+date(value.firstAuthorizedAt)+' · 마지막 인증 '+date(value.lastAuthenticatedAt)+' · 접근 관측 '+date(value.lastObservedAt)+' · 관측 수신 '+date(value.observationReceivedAt)+' · 지연 '+(value.observationDelayMs??'미확인')+'ms · 위임 허가 만료 '+date(value.delegatedGrantExpiresAt);$('detail').hidden=false;$('reason').value='';$('result').textContent='';}catch(error){if(current===generation)fail(error);}}
async function list(more=false){const current=++generation;user=null;pending=null;$('detail').hidden=true;if(!more)$('users').replaceChildren();try{const params=new URLSearchParams({state:$('filter').value,q:$('search').value});if(more&&cursor)params.set('cursor',cursor);const data=await api(base()+'/users?'+params);if(current!==generation)return;cursor=data.nextCursor;$('more').hidden=!cursor;$('summary').textContent='최근 30일 익명 인증 실패 '+data.anonymousAttempts+'회 · 마지막 관측은 온라인 상태가 아닙니다.';if(!data.users.length&&!more)$('users').append(el('p','이 조건에 해당하는 사용자가 없습니다.'));for(const value of data.users){const row=el('div','');row.className='event-row';const button=el('button',(value.displayName||'사전등록 사용자')+' · '+value.discordId+' · '+labels[value.state]);button.className='ghost';button.type='button';button.addEventListener('click',()=>void show(value.subject));row.append(button,el('p','마지막 인증 '+date(value.lastAuthenticatedAt)+' · 접근 관측 '+date(value.lastObservedAt)));$('users').append(row);}status('사용자 목록을 불러왔습니다.');}catch(error){if(current===generation)fail(error);}}
async function service(){const current=++generation,target=selected;user=null;pending=null;capabilities=null;$('detail').hidden=true;try{const value=await api(base()+'/capabilities');if(current!==generation||target!==selected)return;capabilities=value;const option=$('action').querySelector('[value="grant"]');option?.remove();if(value.grantableConditions.includes('additional-roles')){const grant=el('option','추가 역할 조건 임시 허가');grant.value='grant';$('action').append(grant);}$('expiry-field').hidden=true;await list();}catch(error){if(current===generation&&target===selected)fail(error);}}
async function load(){++generation;$('workspace').hidden=true;try{const data=await api('/developer/v1/apps');$('service').replaceChildren();for(const app of data.apps){const option=el('option',app.name+' · '+app.client_id);option.value=app.client_id;$('service').append(option);}if(!data.apps.length){status('관리할 수 있는 서비스가 없습니다.');return;}const requested=new URLSearchParams(location.search).get('client_id');selected=data.apps.some(a=>a.client_id===requested)?requested:data.apps[0].client_id;$('service').value=selected;$('workspace').hidden=false;$('login').hidden=true;await service();}catch(error){fail(error);}}
$('login').addEventListener('click',async()=>{try{sessionStorage.setItem('nakwol.developer.users.return',selected);await auth.login({reauthenticate:true});}catch(error){fail(error);}});
$('service').addEventListener('change',()=>{selected=$('service').value;void service();});$('filter').addEventListener('change',()=>void list());$('reload').addEventListener('click',()=>void list());$('more').addEventListener('click',()=>void list(true));
$('action').addEventListener('change',()=>{$('expiry-field').hidden=$('action').value!=='grant';if($('action').value==='grant'){$('expiry').value=new Date(Date.now()+600000-new Date().getTimezoneOffset()*60000).toISOString().slice(0,16);}pending=null;});$('action-form').addEventListener('input',()=>pending=null);
$('action-form').addEventListener('submit',async event=>{event.preventDefault();if(busy||!user)return;const target=selected,subject=user.subject,action=$('action').value,reason=$('reason').value.trim();if(reason.length<3)return;if(action==='grant'&&!capabilities?.grantableConditions.includes('additional-roles'))return;
const body=pending||{action,reason,expectedVersion:user.version,idempotencyKey:crypto.randomUUID(),...(action==='grant'?{conditions:['additional-roles'],expiresAt:new Date($('expiry').value).getTime()}:{})};
if(!confirm('서비스 '+selected+' / '+user.discordId+'에 '+$('action').selectedOptions[0].textContent+' 조치를 적용합니다. 사유: '+reason))return;pending=body;busy=true;editable();try{const result=await api(base()+'/users/'+subject+'/actions',body);pending=null;if(target!==selected)return;$('result').textContent='조치 저장: '+result.status+' · 전파: '+result.deliveryStatus+' · 로컬 검증 최대 '+result.localLeaseMaxSeconds+'초 · 빠른 전파 지원 미확인. 전체 적용 완료를 뜻하지 않습니다.';const text=$('result').textContent;await list();if(action!=='delete-relationship'){await show(subject);$('result').textContent=text;}else status(text);}catch(error){fail(error);}finally{busy=false;editable();}});
$('preregister').addEventListener('submit',async event=>{event.preventDefault();if(busy)return;busy=true;editable();try{await api(base()+'/users',{discordId:$('discord').value.trim(),reason:$('register-reason').value.trim(),expectedVersion:0,idempotencyKey:crypto.randomUUID()});$('preregister').reset();await list();status('사전등록했습니다. 접근 허가는 별도 조건을 따릅니다.');}catch(error){fail(error);}finally{busy=false;editable();}});
void load();

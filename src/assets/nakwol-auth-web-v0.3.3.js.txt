import { NakwolAuthClient as NakwolAuthClientV032, NakwolAuthError } from '../v0.3.2/nakwol-auth-web.js';
export { DEFAULT_NAKWOL_AUTH_ORIGIN, NakwolAuthError, mountNakwolAuthWidget, mountNakwolIdentityMenu } from '../v0.3.2/nakwol-auth-web.js';
export const NAKWOL_AUTH_SDK_VERSION = '0.3.3';

function boundedAuthFetch(nativeFetch) {
  return async (url, options = {}) => {
    const controller = new AbortController();
    const callerAbort = () => controller.abort(options.signal?.reason);
    const timeoutError = new NakwolAuthError('AUTH_REQUEST_TIMEOUT', '인증 서버 응답이 지연되고 있습니다. 다시 시도하거나 계정 페이지에서 확인해 주세요.');
    let rejectAbort;
    const aborted = new Promise((_, reject) => { rejectAbort = () => reject(controller.signal.reason); });
    controller.signal.addEventListener('abort', rejectAbort, { once: true });
    if (options.signal?.aborted) callerAbort();
    else options.signal?.addEventListener('abort', callerAbort, { once: true });
    const timer = setTimeout(() => controller.abort(timeoutError), 15000);
    let reader;
    try {
      const response = await Promise.race([nativeFetch(url, { ...options, signal: controller.signal }), aborted]);
      const chunks = []; let size = 0;
      if (response.body) {
        reader = response.body.getReader();
        for (;;) {
          const part = await Promise.race([reader.read(), aborted]);
          if (part.done) break;
          size += part.value.byteLength;
          if (size > 65536) throw new NakwolAuthError('AUTH_RESPONSE_TOO_LARGE', '인증 서버 응답이 올바르지 않습니다.');
          chunks.push(part.value);
        }
      }
      const bytes = new Uint8Array(size); let offset = 0;
      for (const chunk of chunks) { bytes.set(chunk, offset); offset += chunk.byteLength; }
      return new Response([204, 205, 304].includes(response.status) ? null : bytes, {
        status: response.status, statusText: response.statusText, headers: response.headers,
      });
    } catch (error) {
      if (reader) reader.cancel().catch(() => {});
      if (controller.signal.aborted) throw controller.signal.reason;
      throw error;
    } finally {
      clearTimeout(timer);
      controller.signal.removeEventListener('abort', rejectAbort);
      options.signal?.removeEventListener('abort', callerAbort);
    }
  };
}

export class NakwolAuthClient extends NakwolAuthClientV032 {
  constructor(options = {}) {
    super({ ...options, fetchImpl: boundedAuthFetch(options.fetchImpl || window.fetch.bind(window)) });
  }
}

export const NAKWOL_AUTH_SDK_VERSION = '0.2.0';
export const DEFAULT_NAKWOL_AUTH_ORIGIN = 'https://nakwol-auth.sepsd21.workers.dev';

export class NakwolAuthError extends Error {
  constructor(code, message, details = null) {
    super(message || code);
    this.name = 'NakwolAuthError';
    this.code = code;
    this.details = details;
  }
}

function base64Url(bytes) {
  let binary = '';
  for (const byte of bytes) binary += String.fromCharCode(byte);
  return btoa(binary).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/g, '');
}

function randomValue(length = 32) {
  const bytes = new Uint8Array(length);
  crypto.getRandomValues(bytes);
  return base64Url(bytes);
}

async function sha256Base64Url(value) {
  const data = new TextEncoder().encode(value);
  const digest = await crypto.subtle.digest('SHA-256', data);
  return base64Url(new Uint8Array(digest));
}

function defaultRedirectUri() {
  return `${location.origin}${location.pathname}`;
}

function assertBrowser() {
  if (typeof window === 'undefined' || typeof document === 'undefined') {
    throw new NakwolAuthError('BROWSER_REQUIRED', 'NAKWOL AUTH Web SDK는 브라우저 환경에서 사용해야 합니다.');
  }
}

export class NakwolAuthClient extends EventTarget {
  constructor(options = {}) {
    super();
    assertBrowser();
    const {
      clientId,
      redirectUri = defaultRedirectUri(),
      authOrigin = DEFAULT_NAKWOL_AUTH_ORIGIN,
      storage = window.sessionStorage,
      fetchImpl = window.fetch.bind(window),
    } = options;

    if (!clientId) throw new NakwolAuthError('CLIENT_ID_REQUIRED', 'clientId가 필요합니다.');
    if (!redirectUri) throw new NakwolAuthError('REDIRECT_URI_REQUIRED', 'redirectUri가 필요합니다.');

    this.clientId = clientId;
    this.redirectUri = redirectUri;
    this.authOrigin = authOrigin.replace(/\/$/, '');
    this.storage = storage;
    this.fetchImpl = fetchImpl;
    this.user = null;
    this.keyPrefix = `nakwol.auth.${clientId}.`;
  }

  key(name) {
    return `${this.keyPrefix}${name}`;
  }

  emit(type, detail = null) {
    this.dispatchEvent(new CustomEvent(type, { detail }));
  }

  getAccessToken() {
    const raw = this.storage.getItem(this.key('token'));
    if (!raw) return null;
    try {
      const record = JSON.parse(raw);
      if (!record.accessToken || !record.expiresAt || Date.now() >= record.expiresAt) {
        this.storage.removeItem(this.key('token'));
        return null;
      }
      return record.accessToken;
    } catch {
      this.storage.removeItem(this.key('token'));
      return null;
    }
  }

  setAccessToken(accessToken, expiresIn) {
    const safetySeconds = 30;
    const ttl = Math.max(1, Number(expiresIn || 3600) - safetySeconds) * 1000;
    this.storage.setItem(this.key('token'), JSON.stringify({
      accessToken,
      expiresAt: Date.now() + ttl,
    }));
  }

  clearLocalState() {
    this.storage.removeItem(this.key('token'));
    this.storage.removeItem(this.key('transaction'));
    this.user = null;
  }

  cleanCallbackUrl() {
    const url = new URL(location.href);
    for (const key of ['code', 'state', 'error', 'error_description']) url.searchParams.delete(key);
    history.replaceState({}, document.title, `${url.pathname}${url.search}${url.hash}`);
  }

  async login() {
    const verifier = randomValue(48);
    const state = randomValue(24);
    const challenge = await sha256Base64Url(verifier);
    this.storage.setItem(this.key('transaction'), JSON.stringify({ verifier, state, createdAt: Date.now() }));

    const url = new URL(`${this.authOrigin}/authorize`);
    url.searchParams.set('response_type', 'code');
    url.searchParams.set('client_id', this.clientId);
    url.searchParams.set('redirect_uri', this.redirectUri);
    url.searchParams.set('code_challenge', challenge);
    url.searchParams.set('code_challenge_method', 'S256');
    url.searchParams.set('state', state);
    this.emit('loginstart', { url: url.toString() });
    location.assign(url.toString());
  }

  async handleCallback() {
    const params = new URLSearchParams(location.search);
    const error = params.get('error');
    const code = params.get('code');
    const state = params.get('state');

    if (error) {
      this.cleanCallbackUrl();
      this.storage.removeItem(this.key('transaction'));
      const authError = new NakwolAuthError(error, params.get('error_description') || 'NAKWOL AUTH 인증이 취소되거나 실패했습니다.');
      this.emit('error', authError);
      throw authError;
    }
    if (!code) return false;

    let transaction = null;
    try {
      transaction = JSON.parse(this.storage.getItem(this.key('transaction')) || 'null');
    } catch {
      transaction = null;
    }

    if (!transaction?.verifier || !transaction?.state || state !== transaction.state) {
      this.cleanCallbackUrl();
      this.storage.removeItem(this.key('transaction'));
      const authError = new NakwolAuthError('STATE_OR_PKCE_MISMATCH', '로그인 요청 상태 또는 PKCE 검증 정보가 일치하지 않습니다.');
      this.emit('error', authError);
      throw authError;
    }

    const response = await this.fetchImpl(`${this.authOrigin}/token?client_id=${encodeURIComponent(this.clientId)}`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        grant_type: 'authorization_code',
        code,
        client_id: this.clientId,
        redirect_uri: this.redirectUri,
        code_verifier: transaction.verifier,
      }),
    });
    const payload = await response.json().catch(() => ({}));
    this.cleanCallbackUrl();
    this.storage.removeItem(this.key('transaction'));

    if (!response.ok || !payload.access_token) {
      const authError = new NakwolAuthError(payload?.error?.code || 'TOKEN_EXCHANGE_FAILED', payload?.error?.message || '인증 코드를 access token으로 교환하지 못했습니다.', payload);
      this.emit('error', authError);
      throw authError;
    }

    this.setAccessToken(payload.access_token, payload.expires_in);
    this.emit('token', { expiresIn: payload.expires_in });
    return true;
  }

  async getMe() {
    const token = this.getAccessToken();
    if (!token) {
      this.user = null;
      return null;
    }

    const response = await this.fetchImpl(`${this.authOrigin}/me?client_id=${encodeURIComponent(this.clientId)}`, {
      headers: { Authorization: `Bearer ${token}` },
    });
    const payload = await response.json().catch(() => ({}));

    if (response.status === 401) {
      this.clearLocalState();
      this.emit('logout', null);
      return null;
    }
    if (!response.ok || !payload?.ok || !payload?.data) {
      const authError = new NakwolAuthError(payload?.error?.code || 'ME_FAILED', payload?.error?.message || '사용자 정보를 불러오지 못했습니다.', payload);
      this.emit('error', authError);
      throw authError;
    }

    this.user = payload.data;
    this.emit('user', this.user);
    return this.user;
  }

  async bootstrap() {
    this.emit('loading', null);
    await this.handleCallback();
    const user = await this.getMe();
    this.emit('ready', user);
    return user;
  }

  isAuthenticated() {
    return Boolean(this.getAccessToken());
  }

  isMember() {
    return Boolean(this.user?.membership?.is_member);
  }

  async logout(options = {}) {
    const { global = false, returnTo = location.href } = options;
    const token = this.getAccessToken();
    if (token) {
      try {
        await this.fetchImpl(`${this.authOrigin}/logout?client_id=${encodeURIComponent(this.clientId)}`, {
          method: 'POST',
          headers: { Authorization: `Bearer ${token}` },
        });
      } catch {
        // Local logout must still succeed if token revocation is temporarily unavailable.
      }
    }

    this.clearLocalState();
    this.emit('logout', null);

    if (global) {
      const url = new URL(`${this.authOrigin}/session/logout`);
      url.searchParams.set('client_id', this.clientId);
      url.searchParams.set('return_to', returnTo);
      location.assign(url.toString());
    }
  }
}

function node(tag, className, text) {
  const element = document.createElement(tag);
  if (className) element.className = className;
  if (text !== undefined && text !== null) element.textContent = text;
  return element;
}

export function mountNakwolAuthWidget(client, options = {}) {
  if (!(client instanceof NakwolAuthClient)) throw new NakwolAuthError('INVALID_CLIENT', 'NakwolAuthClient 인스턴스가 필요합니다.');
  ensureWidgetStyles();

  const root = options.container || node('div', 'nakwol-auth-widget');
  if (!options.container) document.body.appendChild(root);

  function clear() {
    while (root.firstChild) root.removeChild(root.firstChild);
  }

  function card() {
    const value = node('div', 'nakwol-auth-widget__card');
    root.appendChild(value);
    return value;
  }

  function renderLoading() {
    clear();
    const c = card();
    c.appendChild(node('div', 'nakwol-auth-widget__name', options.loadingText || '낙월 인증 확인 중…'));
  }

  function renderLoggedOut(errorMessage = '') {
    clear();
    const c = card();
    const copy = node('div', 'nakwol-auth-widget__copy');
    if (errorMessage) {
      copy.appendChild(node('span', 'nakwol-auth-widget__error', errorMessage));
    } else {
      copy.appendChild(node('span', 'nakwol-auth-widget__name', options.title || 'NAKWOL ID'));
      copy.appendChild(node('span', 'nakwol-auth-widget__meta', options.loggedOutText || '로그인하지 않음'));
    }
    const button = node('button', '', options.loginText || '낙월 로그인');
    button.type = 'button';
    button.addEventListener('click', () => client.login());
    c.append(copy, button);
  }

  function renderUser(user) {
    clear();
    const c = card();
    if (user?.avatar_url) {
      const avatar = node('img', 'nakwol-auth-widget__avatar');
      avatar.src = user.avatar_url;
      avatar.alt = '';
      c.appendChild(avatar);
    } else {
      c.appendChild(node('div', 'nakwol-auth-widget__avatar'));
    }

    const copy = node('div', 'nakwol-auth-widget__copy');
    copy.appendChild(node('span', 'nakwol-auth-widget__name', user?.display_name || user?.id || 'NAKWOL ID'));
    const member = Boolean(user?.membership?.is_member);
    const role = user?.membership?.role || 'user';
    copy.appendChild(node('span', 'nakwol-auth-widget__meta', member ? `인증됨 · ${role}` : `비회원 · ${role}`));

    const button = node('button', '', options.logoutText || '로그아웃');
    button.type = 'button';
    button.dataset.secondary = 'true';
    button.addEventListener('click', () => client.logout({ global: Boolean(options.globalLogout) }));
    c.append(copy, button);
  }

  client.addEventListener('loading', renderLoading);
  client.addEventListener('user', (event) => renderUser(event.detail));
  client.addEventListener('logout', () => renderLoggedOut());
  client.addEventListener('error', (event) => renderLoggedOut(`인증 실패: ${event.detail?.code || 'unknown'}`));

  renderLoading();
  const ready = client.bootstrap().then((user) => {
    if (user) renderUser(user);
    else renderLoggedOut();
    return user;
  }).catch((error) => {
    renderLoggedOut(`인증 실패: ${error?.code || 'unknown'}`);
    throw error;
  });

  return {
    element: root,
    ready,
    refresh: async () => {
      renderLoading();
      const user = await client.getMe();
      if (user) renderUser(user); else renderLoggedOut();
      return user;
    },
    destroy: () => root.remove(),
  };
}

function identityRoleLabel(user) {
  const role = user?.membership?.role || 'user';
  if (role === 'admin') return '낙월 관리자';
  if (user?.membership?.is_member || role === 'member') return '낙월 맹원';
  return '일반 사용자';
}

export function mountNakwolIdentityMenu(client, options = {}) {
  if (!(client instanceof NakwolAuthClient)) throw new NakwolAuthError('INVALID_CLIENT', 'NakwolAuthClient 인스턴스가 필요합니다.');

  const variant = options.variant || 'compact';
  const theme = options.theme || 'inherit';
  if (!IDENTITY_VARIANTS.has(variant)) throw new NakwolAuthError('INVALID_VARIANT', `지원하지 않는 Identity Menu variant입니다: ${variant}`);
  if (!IDENTITY_THEMES.has(theme)) throw new NakwolAuthError('INVALID_THEME', `지원하지 않는 Identity Menu theme입니다: ${theme}`);

  ensureIdentityStyles();
  const root = options.container || node('div', 'nakwol-identity');
  root.classList.add('nakwol-identity');
  root.dataset.variant = variant;
  root.dataset.theme = theme;
  if (!options.container) document.body.appendChild(root);

  const showName = options.showName !== false;
  const showRole = options.showRole !== false;
  let trigger = null;
  let menu = null;
  let destroyed = false;

  const clear = () => {
    while (root.firstChild) root.removeChild(root.firstChild);
    trigger = null;
    menu = null;
  };

  const accountHref = (hash = '') => {
    const url = new URL(options.accountUrl || `${client.authOrigin}/account`);
    if (!options.accountUrl) url.searchParams.set('client_id', client.clientId);
    if (hash) url.hash = hash;
    return url.toString();
  };

  const setOpen = (open, returnFocus = false) => {
    if (!trigger || !menu) return;
    trigger.setAttribute('aria-expanded', open ? 'true' : 'false');
    menu.hidden = !open;
    if (!open && returnFocus) trigger.focus();
  };

  const renderLoading = () => {
    if (destroyed) return;
    clear();
    root.appendChild(node('span', 'nakwol-identity__status', options.loadingText || '낙월 인증 확인 중…'));
  };

  const renderLoggedOut = (errorMessage = '') => {
    if (destroyed) return;
    clear();
    if (errorMessage) {
      const error = node('span', 'nakwol-identity__status', errorMessage);
      error.dataset.error = 'true';
      root.appendChild(error);
    }
    const login = node('button', 'nakwol-identity__button', options.loginText || '낙월 로그인');
    login.type = 'button';
    login.addEventListener('click', () => client.login());
    root.appendChild(login);
  };

  const renderUser = (user) => {
    if (destroyed) return;
    clear();

    trigger = node('button', 'nakwol-identity__trigger');
    trigger.type = 'button';
    trigger.setAttribute('aria-haspopup', 'menu');
    trigger.setAttribute('aria-expanded', 'false');
    trigger.setAttribute('aria-label', '낙월 계정 메뉴 열기');

    if (user?.avatar_url) {
      const avatar = node('img', 'nakwol-identity__avatar');
      avatar.src = user.avatar_url;
      avatar.alt = '';
      trigger.appendChild(avatar);
    } else {
      const fallback = node('span', 'nakwol-identity__avatar', String(user?.display_name || user?.id || 'N').slice(0, 1).toUpperCase());
      fallback.setAttribute('aria-hidden', 'true');
      trigger.appendChild(fallback);
    }

    if (showName) trigger.appendChild(node('span', 'nakwol-identity__name', user?.display_name || user?.id || 'NAKWOL ID'));
    trigger.appendChild(node('span', 'nakwol-identity__chevron', '▾'));

    menu = node('div', 'nakwol-identity__menu');
    menu.hidden = true;
    menu.setAttribute('role', 'menu');

    const summary = node('div', 'nakwol-identity__summary');
    summary.appendChild(node('div', 'nakwol-identity__summary-name', user?.display_name || user?.id || 'NAKWOL ID'));
    if (showRole) summary.appendChild(node('div', 'nakwol-identity__summary-role', identityRoleLabel(user)));
    menu.appendChild(summary);

    const accountLink = node('a', '', '내 낙월 계정');
    accountLink.href = accountHref();
    accountLink.setAttribute('role', 'menuitem');
    menu.appendChild(accountLink);

    const permissionLink = node('a', '', '이 서비스 권한');
    permissionLink.href = accountHref('permissions');
    permissionLink.setAttribute('role', 'menuitem');
    menu.appendChild(permissionLink);

    const logout = node('button', '', options.logoutText || '로그아웃');
    logout.type = 'button';
    logout.setAttribute('role', 'menuitem');
    logout.addEventListener('click', async () => {
      setOpen(false);
      await client.logout({ global: Boolean(options.globalLogout) });
    });
    menu.appendChild(logout);

    trigger.addEventListener('click', () => setOpen(trigger.getAttribute('aria-expanded') !== 'true'));
    root.append(trigger, menu);
  };

  const onDocumentPointer = (event) => {
    if (!root.contains(event.target)) setOpen(false);
  };
  const onDocumentKey = (event) => {
    if (event.key === 'Escape' && trigger?.getAttribute('aria-expanded') === 'true') setOpen(false, true);
  };
  document.addEventListener('pointerdown', onDocumentPointer);
  document.addEventListener('keydown', onDocumentKey);

  const onLoading = () => renderLoading();
  const onUser = (event) => renderUser(event.detail);
  const onLogout = () => renderLoggedOut();
  const onError = (event) => renderLoggedOut(`인증 실패: ${event.detail?.code || 'unknown'}`);
  client.addEventListener('loading', onLoading);
  client.addEventListener('user', onUser);
  client.addEventListener('logout', onLogout);
  client.addEventListener('error', onError);

  renderLoading();
  const ready = client.bootstrap().then((user) => {
    if (user) renderUser(user);
    else renderLoggedOut();
    return user;
  }).catch((error) => {
    renderLoggedOut(`인증 실패: ${error?.code || 'unknown'}`);
    throw error;
  });

  return {
    element: root,
    ready,
    refresh: async () => {
      renderLoading();
      const user = await client.getMe();
      if (user) renderUser(user);
      else renderLoggedOut();
      return user;
    },
    destroy: () => {
      if (destroyed) return;
      destroyed = true;
      document.removeEventListener('pointerdown', onDocumentPointer);
      document.removeEventListener('keydown', onDocumentKey);
      client.removeEventListener('loading', onLoading);
      client.removeEventListener('user', onUser);
      client.removeEventListener('logout', onLogout);
      client.removeEventListener('error', onError);
      root.remove();
    },
  };
}

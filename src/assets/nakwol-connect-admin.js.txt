function toast(message, bad = false) {
  ui.toast.textContent = message;
  ui.toast.dataset.bad = bad ? 'true' : 'false';
  ui.toast.hidden = false;
  setTimeout(() => { ui.toast.hidden = true; }, 3200);
}

async function api(path, options = {}) {
  const token = auth.getAccessToken();
  const headers = new Headers(options.headers || {});
  if (token) headers.set('Authorization', `Bearer ${token}`);
  if (options.body && !headers.has('Content-Type')) headers.set('Content-Type', 'application/json');
  const response = await fetch(path, { ...options, headers });
  const payload = await response.json().catch(() => ({}));
  if (!response.ok) throw new Error(payload?.error?.message || payload?.error?.code || payload?.error || `HTTP ${response.status}`);
  return payload;
}

function clear(node) {
  while (node.firstChild) node.firstChild.remove();
}

function el(tag, className = '', text = '') {
  const node = document.createElement(tag);
  if (className) node.className = className;
  if (text !== '') node.textContent = text;
  return node;
}

function slug(value) {
  return String(value || '')
    .trim()
    .toLowerCase()
    .replace(/[^a-z0-9가-힣]+/g, '-')
    .replace(/^-+|-+$/g, '')
    .replace(/[가-힣]/g, '')
    .replace(/-+/g, '-')
    .slice(0, 63) || 'nakwol-app';
}

function firstRedirect(app) {
  return Array.isArray(app?.redirect_uris) && app.redirect_uris.length ? app.redirect_uris[0] : (app?.homepage_url || '');
}

function renderGate() {
  clear(ui.gate);
  if (!state.user || !state.session) {
    ui.gate.hidden = true;
    ui.workspace.hidden = true;
    return;
  }

  if (state.session.can_manage) {
    ui.gate.hidden = true;
    ui.workspace.hidden = false;
    return;
  }

  ui.workspace.hidden = true;
  ui.gate.hidden = false;
  const box = el('div', 'gate-card');
  if (state.session.bootstrap_available && state.user.membership?.is_member) {
    box.append(
      el('h2', '', '첫 운영자 등록'),
      el('p', '', '아직 NAKWOL Connect 운영자가 없습니다. 최초 1명의 낙월 맹원이 owner가 될 수 있습니다.')
    );
    const claim = el('button', 'primary', '이 계정을 첫 운영자로 등록');
    claim.addEventListener('click', async () => {
      try {
        await api('/admin/api/bootstrap', { method: 'POST' });
        await loadSession();
        await loadApps();
        toast('첫 운영자 등록 완료');
      } catch (error) { toast(error.message, true); }

  } else {
    box.append(
      el('h2', '', '관리 권한이 없습니다'),
      el('p', '', '이 화면은 AUTH 운영자 전용입니다. 등록한 서비스는 내 서비스 관리에서 설정하세요.')

async function loadSession() {
  state.session = (await api('/admin/api/session')).data;
  renderGate();
}

async function loadApps() {
  const payload = await api('/admin/api/apps');
  state.apps = payload.data || [];
  ui.appCount.textContent = String(state.apps.length);
  renderAppList();
  if (state.selected) {
    const fresh = state.apps.find((app) => app.client_id === state.selected.client_id);
    if (fresh) selectApp(fresh);
  } else if (state.apps.length && !state.creating) {
    selectApp(state.apps[0]);
  }
}

function renderAppList() {
  clear(ui.appList);
  for (const app of state.apps) {
    const button = el('button', 'app-item');
    if (state.selected?.client_id === app.client_id && !state.creating) button.dataset.active = 'true';
    const title = el('span', 'app-item-title', app.name);
    const meta = el('span', 'app-item-meta', app.client_id);
    const badge = el('span', `status status-${app.status}`, app.status);
    button.append(title, meta, badge);
    button.addEventListener('click', () => selectApp(app));
    ui.appList.appendChild(button);
  }
}

function setField(name, value) {
  const node = ui.form.elements.namedItem(name);
  if (node) node.value = value ?? '';
}

function getFormData() {
  const fd = new FormData(ui.form);
  return {
    name: String(fd.get('name') || '').trim(),
    client_id: String(fd.get('client_id') || '').trim(),
    homepage_url: String(fd.get('homepage_url') || '').trim(),
    redirect_uris: String(fd.get('redirect_uris') || '').split(/\r?\n/).map((x) => x.trim()).filter(Boolean),
    framework: String(fd.get('framework') || 'other'),
    access_policy: String(fd.get('access_policy') || 'member'),
    status: String(fd.get('status') || 'active'),
  };
}

function embedCode(app) {
  const redirect = firstRedirect(app);
  return `<script\n  src="${location.origin}/connect/v1.js"\n  data-client-id="${app.client_id}"\n  data-redirect-uri="${redirect}">\n</script>`;
}

function nextCode(app) {
  const redirect = firstRedirect(app);
  return `import Script from 'next/script';\n\n<Script\n  src="${location.origin}/connect/v1.js"\n  data-client-id="${app.client_id}"\n  data-redirect-uri="${redirect}"\n  strategy="afterInteractive"\n/>`;
}

function guideFor(app) {
  const framework = app.framework || 'other';
  const generic = embedCode(app);
  const guides = {
    html: ['index.html', '</body> 바로 위', generic],
    vite: ['프로젝트 루트/index.html', '</body> 바로 위', generic],
    react: ['프로젝트 루트/index.html', '</body> 바로 위', generic],
    vue: ['프로젝트 루트/index.html', '</body> 바로 위', generic],
    cra: ['public/index.html', '</body> 바로 위', generic],
    sveltekit: ['src/app.html', '</body> 바로 위', generic],
    next_app: ['app/layout.tsx', '<body> 내부의 마지막 부분', nextCode(app)],
    next_pages: ['pages/_app.tsx', '컴포넌트 반환 JSX 내부', nextCode(app)],
    internal: ['-', '-', generic],
    other: ['사이트의 공통 HTML/레이아웃 파일', '</body> 바로 위', generic],
  };
  return guides[framework] || guides.other;
}

function addCodeBlock(parent, code) {
  const wrap = el('div', 'code-wrap');
  const pre = el('pre', 'code');
  pre.textContent = code;
  const copy = el('button', 'copy', '복사');
  copy.type = 'button';
  copy.addEventListener('click', async () => {
    await navigator.clipboard.writeText(code);
    copy.textContent = '복사됨';
    setTimeout(() => { copy.textContent = '복사'; }, 1200);
  });
  wrap.append(pre, copy);
  parent.appendChild(wrap);
}

function renderDiagnostics(data) {
  clear(ui.diagnostics);
  if (!state.selected) {
    ui.diagnostics.appendChild(el('p', 'muted', '저장된 앱을 선택하세요.'));
    return;
  }
  if (!data) {
    ui.diagnostics.appendChild(el('p', 'muted', '“연동 상태 확인”을 누르면 실제 서비스 주소를 검사합니다.'));
    return;
  }
  ui.diagnostics.append(
    statusRow('앱 등록', data.registered, data.status || ''),
    statusRow('Redirect URI', data.redirect_ok, data.redirect_uri || ''),
    statusRow('서비스 응답', data.site_reachable, data.http_status ? `HTTP ${data.http_status}` : data.error || ''),
    statusRow('Connect 스크립트', data.embed_found, data.embed_found ? 'connect/v1.js 발견' : '페이지 HTML에서 찾지 못함'),
    statusRow('Client ID', data.client_id_found, data.client_id_found ? state.selected.client_id : '페이지 HTML에서 찾지 못함')
  );
}

async function runDiagnostics() {
  if (!state.selected) return;
  ui.diagButton.disabled = true;
  ui.diagButton.textContent = '확인 중…';
  try {
    const payload = await api(`/admin/api/apps/${encodeURIComponent(state.selected.client_id)}/diagnostics`);
    renderDiagnostics(payload.data);
  } catch (error) {
    renderDiagnostics({ registered: true, redirect_ok: true, site_reachable: false, embed_found: false, client_id_found: false, error: error.message });
  } finally {
    ui.diagButton.disabled = false;
    ui.diagButton.textContent = '연동 상태 확인';
  }
}

async function loadIssues(append = false) {
  if (!state.selected) return;
  const appId = state.selected.client_id, generation = ++issuesGeneration;
  const host = $('#support-issues'), more = $('#issues-more');
  more.hidden = true;
  if (!append) { issuesCursor = null; host.textContent = '접근 거부 사용자의 현재 권한을 확인하고 있습니다…'; }
  try {
    const data = await api('/admin/api/access/' + encodeURIComponent(appId) + '/issues' +
      (append && issuesCursor ? '?after=' + encodeURIComponent(issuesCursor) : ''));
    if (generation !== issuesGeneration || state.selected?.client_id !== appId) return;
    if (!append) clear(host);
    if (!data.issues.length && !append) host.append(el('p', 'muted', '기록된 접근 거부 사용자가 없습니다.'));
    for (const issue of data.issues) {
      const row = el('div', 'event-row'), d = issue.diagnosis;
      row.append(el('b', '', issue.display_name + ' · ' + (d.allowed ? '현재 접근 가능' : '조치 필요')),
        el('code', '', 'Discord ' + (issue.discord_user_id || '미확인')),
        el('p', '', reasonLabels[d.reason] || d.reason),
        el('p', 'muted', '마지막 거부: ' + formatTime(issue.last_failed_at) + ' · 기록 ' + issue.failure_count + '회'));
      if (issue.discord_user_id) {
        const inspect = el('button', 'ghost', '상세 진단'); inspect.type = 'button';
        inspect.onclick = () => { supportId.value = issue.discord_user_id; loadEvents().catch(showSupportError); };
        row.append(inspect);
        if (!d.allowed && ['SEASON_ROLE_MISSING', 'ADDITIONAL_ROLE_MISSING', 'MEMBERSHIP_INACTIVE', 'MEMBERSHIP_REFRESH_REQUIRED', 'QUALIFICATION_MISSING'].includes(d.reason) && ['member','guest'].includes(d.policy)) {
          const grant = el('button', 'primary', '이 서비스 접근 허가'); grant.type = 'button';
          grant.onclick = async () => { grant.disabled = true; try { await actOnAccess('grant', issue.discord_user_id); } finally { grant.disabled = false; } };
          row.append(grant);
        }
      }
      host.append(row);
    }
    issuesCursor = data.next_cursor; more.hidden = !data.has_more;
  } catch (error) {
    if (generation !== issuesGeneration || state.selected?.client_id !== appId) return;
    if (!append) clear(host);
    host.append(el('p', 'error-text', '문제 사용자 조회 실패: ' + error.message));
  }
}

function renderEvents(events, append = false) {

  if (!events.length && !append) ui.events.appendChild(el('p', 'muted', '기록된 인증 이벤트가 없습니다.'));
  for (const event of events) {
    const detail = readDetail(event.detail);
    const row = el('div', 'event-row');
    const top = el('div', 'event-top');
    top.append(el('b', '', event.event_type), el('time', '', formatTime(event.created_at)));
    row.append(top, el('p', '', event.display_name || '사용자 미확인'),
      el('code', '', 'Discord ' + (event.discord_user_id || detail.discord_user_id || '미확인') + ' · ' + (event.user_id || '-')));
    if (detail.diagnosis) row.append(el('p', '', reasonLabels[detail.diagnosis.reason] || detail.diagnosis.reason));
    if (detail.reason) row.append(el('p', '', '조치 사유: ' + detail.reason));
    if (detail.actor_user_id) row.append(el('code', '', '조치 운영자: ' + detail.actor_user_id));
    if (detail.message) row.append(el('p', 'error-text', detail.message));
    const discordId = event.discord_user_id || detail.discord_user_id;
    if (discordId) {
      const select = el('button', 'ghost', '이 계정 진단');
      select.type = 'button';
      select.onclick = () => { supportId.value = discordId; loadEvents().catch(showSupportError); };
      row.append(select);
    }
    ui.events.appendChild(row);
  }
}
function renderEventsError(message) { clear(ui.events); ui.events.appendChild(el('p', 'error-text', message)); }
function showSupportError(error) { supportStatus.textContent = error.message; supportStatus.className = 'error-text'; }
async function loadEvents(append = false) {
  if (!state.selected) return;
  const generation = ++supportGeneration;
  const params = new URLSearchParams({ discord_id: supportId.value.trim() });
  if (append === true && eventsCursor) { params.set('before', eventsCursor.created_at); params.set('cursor_id', eventsCursor.id); }
  const payload = await api('/admin/api/access/' + encodeURIComponent(state.selected.client_id) + '?' + params);
  if (generation !== supportGeneration) return;
  renderEvents(payload.events || [], append === true);
  eventsCursor = payload.events?.at(-1) || null;
  moreEvents.hidden = !payload.has_more;
  const diagnosis = $('#support-diagnosis'); clear(diagnosis);
  if (payload.diagnosis) {
    const d = payload.diagnosis;
    diagnosis.append(el('h4', '', payload.user.display_name + ' · ' + (d.allowed ? '접근 허용' : '접근 거부')),
      el('p', '', reasonLabels[d.reason] || d.reason),
      el('p', 'muted', 'Discord 역할 마지막 확인: ' + formatTime(d.checked_at)),
      el('p', 'muted', '부족한 역할 ID: ' + (d.missing_role_ids.join(', ') || '없음')),
      el('p', 'muted', '재인증 요구: ' + formatTime(d.reauthentication_requested_at) + (d.reauthentication_status === 'completed' ? ' · 재확인 완료' : d.reauthentication_status === 'pending' ? ' · 재확인 대기' : '')));
  } else if (supportId.value.trim()) diagnosis.append(el('p', 'muted', '아직 로그인한 적 없는 Discord ID입니다. 수동 허가는 미리 등록할 수 있습니다.'));
  const grants = $('#support-grants'); clear(grants);
  if (!payload.grants.length) grants.append(el('p', 'muted', '수동 허가 기록이 없습니다.'));
  for (const grant of payload.grants) {
    const row = el('div', 'event-row');
    row.append(el('b', '', (grant.display_name || '가입 전 사용자') + ' · ' + (grant.status === 'active' ? '허가 중' : '회수됨')),
      el('code', '', grant.discord_user_id), el('p', 'muted', grant.reason + ' · ' + formatTime(grant.updated_at)));
    const select = el('button', 'ghost', '계정 선택'); select.type = 'button';
    select.onclick = () => { supportId.value = grant.discord_user_id; loadEvents().catch(showSupportError); };
    row.append(select); grants.append(row);
  }
}

async function actOnAccess(action, selectedDiscordId) {

const usageNumber = new Intl.NumberFormat('ko-KR');
const usageTime = new Intl.DateTimeFormat('ko-KR', { month: 'numeric', day: 'numeric', hour: '2-digit', minute: '2-digit', hourCycle: 'h23' });
function usageTile(label, value, note) {
  const tile = el('div', 'usage-tile');
  tile.append(el('span', '', label), el('b', '', usageNumber.format(value)));
  if (note) tile.append(el('small', '', note));
  return tile;
}
function usageChart(daily) {
  const peak = Math.max(1, ...daily.map(day => day.users));
  const chart = el('div', 'usage-chart');
  chart.setAttribute('role', 'img');
  chart.setAttribute('aria-label', '일별 이용자 수. 최고 ' + peak + '명');
  for (const day of daily) {
    const bar = el('div', day.users ? 'usage-bar' : 'usage-bar idle');
    bar.style.height = day.users ? Math.max(4, Math.round(day.users / peak * 100)) + '%' : '2px';
    bar.title = day.date + ' · 이용자 ' + day.users + '명 · 세션 ' + day.sessions + '회';
    chart.append(bar);
  }
  const axis = el('div', 'usage-axis');
  axis.append(el('span', '', daily[0]?.date || ''), el('span', '', '일별 이용자 수 (최고 ' + peak + '명)'), el('span', '', daily.at(-1)?.date || ''));
  return [chart, axis];
}
const usageColumns = [
  ['rank', '#', true],
  ['name', '사용자', false],
  ['sessions', '이용 세션', true],
  ['active_days', '이용한 날', true],
  ['last_used_at', '마지막 이용', false],
  ['first_used_at', '기간 내 첫 이용', false],
];
function usageSorted(users) {
  const ranked = users.map((user, index) => ({ ...user, rank: index + 1 }));
  if (usageSort === 'sessions') return ranked;
  return ranked.sort((a, b) => (b[usageSort] - a[usageSort]) || (a.rank - b.rank));
}
function usageTable(data) {
  const wrap = el('div', 'usage-table-wrap');
  const table = el('table', 'usage-table');
  const head = el('tr');
  for (const [key, label, numeric] of usageColumns) {
    const th = el('th', numeric ? 'num' : '');
    th.scope = 'col';
    if (['sessions', 'active_days', 'last_used_at'].includes(key)) {
      const button = el('button', '', label);
      button.type = 'button';
      if (usageSort === key) button.setAttribute('aria-sort', 'descending');
      button.onclick = () => { usageSort = key; renderUsage(data); };
      th.append(button);
    } else th.textContent = label;
    head.append(th);
  }
  const thead = el('thead'); thead.append(head);
  const tbody = el('tbody');
  for (const user of usageSorted(data.users)) {
    const row = el('tr');
    const who = el('div', 'usage-user');
    if (user.avatar_url) { const img = el('img'); img.src = user.avatar_url; img.alt = ''; img.loading = 'lazy'; img.referrerPolicy = 'no-referrer'; who.append(img); }
    else who.append(el('span', 'usage-avatar'));
    const name = el('div');
    name.append(el('div', '', user.display_name || '이름 미확인'), el('code', 'muted', user.discord_user_id ? 'Discord ' + user.discord_user_id : user.user_id));
    who.append(name);
    const nameCell = el('td'); nameCell.append(who);
    row.append(el('td', 'num', String(user.rank)), nameCell, el('td', 'num', usageNumber.format(user.sessions)),
      el('td', 'num', user.active_days + '일'), el('td', '', usageTime.format(user.last_used_at)), el('td', '', usageTime.format(user.first_used_at)));
    tbody.append(row);
  }
  table.append(thead, tbody);
  wrap.append(table);
  return wrap;
}
function renderUsage(data) {
  const target = $('#usage-area');
  clear(target);
  const s = data.summary;
  const tiles = el('div', 'usage-tiles');
  tiles.append(usageTile('이용자', s.users, '최근 ' + data.days + '일'), usageTile('이용 세션', s.sessions, s.users ? '1인 평균 ' + (s.sessions / s.users).toFixed(1) + '회' : ''),
    usageTile('접근 거부', s.denied, s.denied_users ? s.denied_users + '명 · 아래 접근 문제 해결 참고' : ''));
  target.append(tiles, ...usageChart(data.daily));
  if (!data.users.length) { target.append(el('p', 'muted', '이 기간에 기록된 이용이 없습니다.')); return; }
  target.append(el('h4', '', '자주 이용한 사용자'), usageTable(data));
  if (data.users_truncated) target.append(el('p', 'muted', '이용 세션 수 상위 ' + data.users.length + '명만 표시합니다.'));
}

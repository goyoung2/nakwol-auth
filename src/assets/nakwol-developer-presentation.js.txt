import {NakwolAuthClient} from '/sdk/v0.3.2/nakwol-auth-web.js';
import {parsePresentation} from '/presentation/v1/presentation-schema.mjs';
const auth=new NakwolAuthClient({clientId:'nakwol-connect-admin',redirectUri:location.origin+'/admin/apps',authOrigin:location.origin});
const $=id=>document.getElementById(id),colors=['background','surface','text','muted','accent'];
let selected='',record=null,draft=null,screen='login',generation=0,busy=false,previewDocument=null,assetUrls=new Map();
function status(text){$('status').textContent=text;}
function base(){return '/developer/v1/apps/'+encodeURIComponent(selected)+'/presentation';}
async function api(path,body,method='POST'){
 const token=auth.getAccessToken();if(!token)throw Object.assign(new Error('관리자 로그인이 필요합니다'),{status:401});
 const response=await fetch(path,{method:body===undefined?'GET':method,cache:'no-store',headers:{Authorization:'Bearer '+token,'Content-Type':'application/json'},body:body===undefined?undefined:JSON.stringify(body)});const payload=await response.json();if(!response.ok)throw Object.assign(new Error(payload.error?.code||'요청 실패'),{status:response.status});return payload.data;
}
function fail(error){status(error.status===409?'버전·권한 또는 업로드 조건이 변경되었습니다. 다시 불러오기 후 확인하세요.':error.status===401||error.status===403?'관리자 재인증 또는 현재 서비스 소유권을 확인하세요.':error.message);if(error.status===401||error.status===403)$('login').hidden=false;}
function lock(value){busy=value;for(const id of ['service','editor','rollback','history','reload']){const el=$(id);if(el.tagName==='FORM')for(const field of el.elements)field.disabled=value;else el.disabled=value;}if(!value)$('upload-button').disabled=!record?.uploadAvailable;}
function resetAssets(){for(const url of assetUrls.values())URL.revokeObjectURL(url);assetUrls.clear();}
function showScreen(){const s=draft.screens[screen];$('service-name').value=s.serviceName;$('screen-title').value=s.title;$('screen-copy').value=s.copy;$('layout').value=s.layout;$('logo').value=s.logoAssetId||'';$('background-image').value=s.backgroundAssetId||'';for(const id of ['service-name','screen-title','screen-copy','logo','background-image'])$(id).disabled=screen==='checking';}
function read(){const s=draft.screens[screen];s.serviceName=$('service-name').value;s.title=$('screen-title').value;s.copy=$('screen-copy').value;s.layout=$('layout').value;s.logoAssetId=$('logo').value||null;s.backgroundAssetId=$('background-image').value||null;const w=draft.widget;for(const key of ['visibility','variant','placement','position'])w[key]=$(key).value;w.margin=Number($('margin').value);w.showName=$('show-name').value==='true';draft.theme.mode=$('mode').value;draft.theme.radius=Number($('radius').value);draft.theme.size=$('size').value;for(const key of colors)draft.theme[key]=$('color-'+key).value;draft.support={url:$('support-url').value.trim()||null,label:$('support-label').value};return parsePresentation(draft);}
function fill(data){record=data;draft=structuredClone(data.draft);screen=$('screen').value;$('versions').textContent='초안 revision '+data.version+' · 게시 버전 '+(data.publishedVersion??'없음');for(const key of ['visibility','variant','placement','position'])$(key).value=draft.widget[key];$('margin').value=draft.widget.margin;$('show-name').value=String(draft.widget.showName);$('mode').value=draft.theme.mode;$('radius').value=draft.theme.radius;$('size').value=draft.theme.size;for(const key of colors)$('color-'+key).value=draft.theme[key];$('support-url').value=draft.support.url||'';$('support-label').value=draft.support.label;
 for(const id of ['logo','background-image','delete-asset']){const select=$(id);select.replaceChildren(new Option('없음',''));for(const asset of data.assets)select.add(new Option(asset.id.slice(0,8)+' · '+asset.width+'×'+asset.height,asset.id));}
 $('history').replaceChildren();for(const version of data.history)$('history').add(new Option('버전 '+version.version+' · '+new Date(version.createdAt).toLocaleString(),String(version.version)));$('upload-button').disabled=!data.uploadAvailable;showScreen();}
async function refresh(){const current=++generation,target=selected;resetAssets();record=null;previewDocument=null;try{const data=await api(base()+'/draft');if(current!==generation||target!==selected)return;fill(data);$('workspace').hidden=false;status('초안을 불러왔습니다. 게시 전 미리보기로 확인하세요.');}catch(error){if(current===generation)fail(error);}}
async function load(){const current=++generation;try{const data=await api('/developer/v1/apps');if(current!==generation)return;$('service').replaceChildren();for(const app of data.apps)$('service').add(new Option(app.name+' · '+app.client_id,app.client_id));if(!data.apps.length){status('관리할 수 있는 서비스가 없습니다.');return;}const requested=new URLSearchParams(location.search).get('client_id');selected=data.apps.some(app=>app.client_id===requested)?requested:data.apps[0].client_id;$('service').value=selected;$('login').hidden=true;await refresh();}catch(error){if(current===generation)fail(error);}}
function reason(){const value=$('reason').value.trim();if(value.length<3)throw new Error('변경 사유를 3자 이상 입력하세요.');return value;}
async function mutation(action,body,method){if(busy||!record)return;lock(true);try{const result=await api(base()+'/'+action,{expectedVersion:record.version,reason:reason(),...body},method);fill({...result,uploadAvailable:record.uploadAvailable});status(action==='draft'?'초안을 저장했습니다. 공개 화면에는 아직 적용되지 않습니다.':'새 브랜드 버전을 게시했습니다. 지원 runtime은 다음 화면 로드 시 갱신하며 로그인 세션은 유지됩니다.');}catch(error){fail(error);}finally{lock(false);showScreen();}}
function sendPreview(){if(!previewDocument)return;const frame=$('preview-frame');frame.style.width=$('viewport').value==='mobile'?'375px':'100%';frame.contentWindow.postMessage({type:'nakwol-presentation-preview',clientId:selected,presentation:previewDocument,state:$('preview-state').value,assets:Object.fromEntries(assetUrls)},location.origin);}
async function preview(){if(busy||!record)return;const current=generation,target=selected;lock(true);try{const data=await api(base()+'/preview',{expectedVersion:record.version,reason:'owner private preview'});if(current!==generation||target!==selected)return;const p=parsePresentation(data.presentation),ids=[...new Set(Object.values(p.screens).flatMap(s=>[s.logoAssetId,s.backgroundAssetId]).filter(Boolean))];for(const id of ids){if(assetUrls.has(id))continue;const asset=await api(base()+'/assets/'+encodeURIComponent(id));if(current!==generation||target!==selected)return;const bytes=Uint8Array.from(atob(asset.base64),c=>c.charCodeAt(0));assetUrls.set(id,URL.createObjectURL(new Blob([bytes],{type:asset.mime})));}previewDocument=p;$('preview-note').textContent='저장한 초안 revision '+data.version+' · 선택한 상태를 재현합니다. 실제 로그인은 실행하지 않습니다.';sendPreview();status('비공개 초안을 미리보고 있습니다.');}catch(error){if(current===generation)fail(error);}finally{lock(false);showScreen();}}
$('mode').addEventListener('change',()=>{if($('mode').value==='system')return;const light=$('mode').value==='light';const palette=light?{background:'#f6f3eb',surface:'#ffffff',text:'#182235',muted:'#475569',accent:'#5865f2'}:{background:'#0b1020',surface:'#111a2b',text:'#f1f5f9',muted:'#a6b3c7',accent:'#5865f2'};for(const key of colors)$('color-'+key).value=palette[key];});
$('login').addEventListener('click',()=>void auth.login({reauthenticate:true}));$('reload').addEventListener('click',()=>void load());$('service').addEventListener('change',()=>{selected=$('service').value;$('workspace').hidden=true;void refresh();});$('screen').addEventListener('change',()=>{read();screen=$('screen').value;showScreen();});$('editor').addEventListener('submit',event=>{event.preventDefault();try{void mutation('draft',{presentation:read()},'PUT');}catch(error){fail(error);}});$('publish').addEventListener('click',()=>{if(confirm('저장한 초안을 이 서비스의 새 공개 브랜드로 게시합니다.'))void mutation('publish',{});});$('rollback').addEventListener('click',()=>{if(confirm('선택한 이전 버전으로 새 브랜드 버전을 게시합니다.'))void mutation('rollback',{targetVersion:Number($('history').value)});});$('preview').addEventListener('click',()=>void preview());$('preview-state').addEventListener('change',sendPreview);$('viewport').addEventListener('change',sendPreview);$('preview-frame').addEventListener('load',sendPreview);
$('upload-button').addEventListener('click',async()=>{if(busy||!record)return;const file=$('upload').files?.[0];if(!file||file.size>524288||!['image/png','image/jpeg','image/webp'].includes(file.type)){status('PNG/JPEG/WebP 파일을 선택하세요. 최대 512KiB입니다.');return;}let why;try{why=reason();read();}catch(error){fail(error);return;}lock(true);try{const bytes=new Uint8Array(await file.arrayBuffer());let binary='';for(let i=0;i<bytes.length;i+=8192)binary+=String.fromCharCode(...bytes.subarray(i,i+8192));const asset=await api(base()+'/assets',{base64:btoa(binary),mime:file.type,reason:why});const data=await api(base()+'/draft');const local=draft;fill({...data,draft:local});$($('upload-target').value).value=asset.id;read();status('이미지를 안전한 WebP로 저장했습니다. 초안 저장 후 미리보기로 확인하세요.');}catch(error){fail(error);}finally{lock(false);showScreen();}});
window.addEventListener('beforeunload',resetAssets);void load();
$('delete-asset-button').addEventListener('click',async()=>{if(busy||!record||!$('delete-asset').value)return;let why;try{why=reason();}catch(error){fail(error);return;}const id=$('delete-asset').value;if(!confirm('초안과 보존된 게시 이력이 사용하지 않는 파일만 삭제합니다. 계속할까요?'))return;lock(true);try{await api(base()+'/assets/'+id,{reason:why,expectedVersion:record.version},'DELETE');resetAssets();await refresh();status('사용하지 않는 브랜드 파일을 삭제했습니다.');}catch(error){fail(error);}finally{lock(false);showScreen();}});

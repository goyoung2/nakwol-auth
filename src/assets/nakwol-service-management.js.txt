import { NakwolAuthClient } from '/sdk/v0.3.2/nakwol-auth-web.js';
const auth = new NakwolAuthClient({clientId:'nakwol-connect-admin',redirectUri:location.origin+'/admin/apps',authOrigin:location.origin});
const $ = id => document.getElementById(id);
const ui = Object.fromEntries(['status','login','reload','workspace','service','capabilities','policy-form','policy-fields','idle','absolute','lease','reason','save','policy-values','application-status','operator-token','token','operator-access','access','preview','preview-text','confirm','cancel-preview'].map(id=>[id,$(id)]));
let selected = '', policy = null, capabilities = null, requestGeneration = 0, saving = false, isOperator = false, pendingPreview = null;
const GLOBAL = '@global';
const endpoint = () => selected === GLOBAL ? '/admin/api/auth-policy' : (isOperator ? '/admin/api/auth-policy/'+encodeURIComponent(selected) : '/developer/v1/apps/'+encodeURIComponent(selected)+'/policy');
function clearPreview() { pendingPreview=null; ui.preview.hidden=true; }
function status(text) { ui.status.textContent = text; }
function editable() { ui['policy-fields'].disabled = saving || !policy || !capabilities?.capabilities.includes('policy:write'); }
async function api(path, options = {}) {
  const token = auth.getAccessToken();
  if (!token) throw Object.assign(new Error('로그인이 필요합니다.'),{code:'UNAUTHENTICATED'});
  const response = await fetch(path,{...options,cache:'no-store',headers:{Authorization:'Bearer '+token,'Content-Type':'application/json'}});
  const payload = await response.json();
  if (!response.ok) throw Object.assign(new Error(payload.error?.message || payload.error?.code || '요청 실패'),{code:payload.error?.code,status:response.status});
  return payload.data;
}
function fail(error) {
  if (error.status === 409) status('다른 관리자가 정책을 변경했습니다. 다시 불러온 뒤 변경 내용을 확인하고 저장하세요.');
  else if (error.status === 401 || /AUTH|REAUTH/.test(error.code || '')) { status('관리자 인증을 갱신한 뒤 다시 시도하세요.'); ui.login.hidden = false; }
  else status(error.message || '처리하지 못했습니다. 다시 불러오기를 눌러 주세요.');
}
function summary(value) { return '미사용 '+value.sessionIdleSeconds/3600+'시간 · 최대 '+value.sessionAbsoluteSeconds/3600+'시간 · 권한 재확인 '+value.leaseSeconds+'초'; }
function renderPolicy(value) {
  policy = value;
  const effective = value.effective;
  ui.idle.value = String(effective.sessionIdleSeconds/3600); ui.absolute.value = String(effective.sessionAbsoluteSeconds/3600); ui.lease.value = String(effective.leaseSeconds); ui.token.value=String(effective.accessTokenSeconds); ui.access.value=value.accessPolicy || 'member';
  const stored = value.stored || {};
  ui['policy-values'].textContent = '정책 버전 '+value.policyVersion+' · 저장값: '+['sessionIdleSeconds','sessionAbsoluteSeconds','leaseSeconds'].map(key=>key+'='+ (stored[key] ?? '전역 기본값')).join(', ')+' · 실효값: '+summary(effective);
  ui['application-status'].textContent = '권한 재확인 간격·접근 정책: 대응 런타임에서 다음 생성 또는 중앙 재검증 시 적용(기존 세션 최대 5분). 세션 미사용·최대 유지 시간: 현재는 저장만 하며 서버 갱신 기능(T06) 적용 대기입니다. 이미 만료된 세션은 기간을 늘려도 복구되지 않습니다.';
  editable();
}
async function loadPolicy() {
  clearPreview(); const generation = ++requestGeneration; policy = null; capabilities = null; editable();
  if (!selected) return;
  status('정책을 불러오는 중…');
  try {
    const global = selected === GLOBAL;
    const [nextPolicy,nextCapabilities] = await Promise.all([api(endpoint()),global ? Promise.resolve({capabilities:['policy:read','policy:write'],ranges:{sessionIdleSeconds:{min:3600,max:864000},sessionAbsoluteSeconds:{min:3600,max:2592000},leaseSeconds:{min:60,max:300}}}) : api('/developer/v1/apps/'+encodeURIComponent(selected)+'/capabilities')]);
    if (generation !== requestGeneration) return;
    capabilities = nextCapabilities; ui['operator-token'].hidden=!isOperator; ui['operator-access'].hidden=!isOperator || global; ui.save.textContent=isOperator ? '변경 영향 확인' : '정책 저장';
    for (const [id,key,divisor] of [['idle','sessionIdleSeconds',3600],['absolute','sessionAbsoluteSeconds',3600],['lease','leaseSeconds',1]]) {
      const range = capabilities.ranges[key]; ui[id].min = String(range.min/divisor); ui[id].max = String(range.max/divisor);
    }
    ui.capabilities.textContent = capabilities.capabilities.includes('policy:write') ? '이 서비스의 세션 정책을 변경할 수 있습니다.' : '이 서비스의 정책을 조회할 수 있습니다. 변경 권한은 없습니다.';
    ui.reason.value = ''; renderPolicy(nextPolicy); status('정책을 불러왔습니다.');
  } catch(error) { if(generation===requestGeneration) fail(error); }
}
async function load() {
  ++requestGeneration; policy=null; capabilities=null; editable(); ui.workspace.hidden=true;
  try {
    const data = await api('/developer/v1/apps');
    isOperator=data.isOperator===true; ui.service.replaceChildren();
    if(isOperator) {const option=document.createElement('option'); option.value=GLOBAL; option.textContent='전역 기본값 · AUTH 운영자'; ui.service.append(option);}
    for (const app of data.apps) { const option = document.createElement('option'); option.value=app.client_id; option.textContent=app.name+' · '+app.client_id; ui.service.append(option); }
    if (!data.apps.length && !isOperator) { status('관리할 수 있는 서비스가 없습니다. AUTH 운영자에게 서비스 소유권 등록을 요청하세요.'); return; }
    const requested = selected || new URLSearchParams(location.search).get('client_id');
    selected = requested===GLOBAL && isOperator ? GLOBAL : data.apps.some(app=>app.client_id===requested) ? requested : data.apps[0]?.client_id || GLOBAL;
    ui.service.value=selected; ui.workspace.hidden=false; ui.login.hidden=true; await loadPolicy();
  } catch(error) { fail(error); }
}
ui.login.addEventListener('click',async()=>{
  try { sessionStorage.setItem('nakwol.developer.return',selected); await auth.login({reauthenticate:true}); } catch(error) { fail(error); }
});
ui.reload.addEventListener('click',()=>{ if(!saving) void load(); });
ui.service.addEventListener('change',()=>{ selected=ui.service.value; void loadPolicy(); });
ui['policy-form'].addEventListener('submit',async event=>{
  event.preventDefault(); if(saving || !policy || !capabilities?.capabilities.includes('policy:write')) return;
  const idle = Number(ui.idle.value)*3600, absolute = Number(ui.absolute.value)*3600, lease = Number(ui.lease.value), reason=ui.reason.value.trim();
  if (!Number.isInteger(idle) || !Number.isInteger(absolute) || !Number.isInteger(lease) || idle>absolute || reason.length<3) { status('미사용 만료는 최대 유지 이하로, 변경 사유는 3자 이상 입력하세요.'); return; }
  const target=selected, version=policy.policyVersion;
  const patch={sessionIdleSeconds:idle,sessionAbsoluteSeconds:absolute,leaseSeconds:lease};
  if(isOperator) {patch.accessTokenSeconds=Number(ui.token.value); if(selected!==GLOBAL && ui.access.value!==policy.accessPolicy) patch.accessPolicy=ui.access.value;}
  const body={expectedVersion:version,patch,reason}; saving=true; editable(); ui.service.disabled=true; ui.reload.disabled=true;
  try {
    if(isOperator) {
      const result=await api(endpoint()+'/preview',{method:'POST',body:JSON.stringify(body)});
      pendingPreview={target,endpoint:endpoint(),body:{...body,previewToken:result.previewToken}};
      ui['preview-text'].textContent='정책 버전 '+version+' → '+summary(result.effective)+' · 토큰 '+result.effective.accessTokenSeconds+'초 · 접근 정책 '+(result.accessPolicy || '서비스별 정책 유지');
      ui.preview.hidden=false; status('변경 영향을 확인한 뒤 저장하세요.'); return;
    }
    await api(endpoint(),{method:'PUT',body:JSON.stringify(body)});
    await loadPolicy(); status('정책을 저장했습니다. 아래 적용 상태와 사이트 런타임 지원 여부를 확인하세요.');
  } catch(error) { if(error.status===409) {policy=null; editable();} fail(error); }
  finally {saving=false; editable(); ui.service.disabled=false; ui.reload.disabled=false;}
});
ui['policy-form'].addEventListener('input',clearPreview);
ui['cancel-preview'].addEventListener('click',clearPreview);
ui.confirm.addEventListener('click',async()=>{
  if(saving || !pendingPreview || pendingPreview.target!==selected) return;
  const preview=pendingPreview; saving=true; editable(); ui.confirm.disabled=true; ui.service.disabled=true; ui.reload.disabled=true;
  try { await api(preview.endpoint,{method:'PUT',body:JSON.stringify(preview.body)}); clearPreview(); await loadPolicy(); status('정책을 저장했습니다. 적용 상태를 확인하세요.'); }
  catch(error) {clearPreview(); if(error.status===409) policy=null; fail(error);}
  finally {saving=false; editable(); ui.confirm.disabled=false; ui.service.disabled=false; ui.reload.disabled=false;}
});
void load();

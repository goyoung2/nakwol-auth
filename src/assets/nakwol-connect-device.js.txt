import { NakwolAuthClient } from 'https://nakwol-auth.sepsd21.workers.dev/sdk/v0.1.0/nakwol-auth-web.js';

const STORAGE_KEY = 'nakwol.connect.device.userCode';
const auth = new NakwolAuthClient({
  clientId: 'nakwol-connect-admin',
  redirectUri: `${location.origin}/connect/device`,
});

const codeEl = document.querySelector('#user-code');
const statusEl = document.querySelector('#status');
const loginButton = document.querySelector('#login');
const approveButton = document.querySelector('#approve');
const denyButton = document.querySelector('#deny');

function currentUserCode() {
  const params = new URLSearchParams(location.search);
  const fromUrl = params.get('user_code');
  if (fromUrl) {
    const normalized = fromUrl.trim().toUpperCase();
    sessionStorage.setItem(STORAGE_KEY, normalized);
    return normalized;
  }
  return sessionStorage.getItem(STORAGE_KEY) || '';
}

function setStatus(text, kind = '') {
  statusEl.textContent = text;
  statusEl.className = kind;
}

async function decide(endpoint) {
  const userCode = currentUserCode();
  const token = auth.getAccessToken();
  if (!userCode || !token) return;
  approveButton.disabled = true;
  denyButton.disabled = true;
  try {
    const response = await fetch(endpoint, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        Authorization: `Bearer ${token}`,
      },
      body: JSON.stringify({ user_code: userCode }),
    });
    const payload = await response.json().catch(() => ({}));
    if (!response.ok || !payload?.ok) {
      throw new Error(payload?.error?.message || payload?.error?.code || `HTTP ${response.status}`);
    }
    sessionStorage.removeItem(STORAGE_KEY);
    approveButton.hidden = true;
    denyButton.hidden = true;
    setStatus(endpoint.endsWith('/approve') ? '승인했습니다. CLI 창으로 돌아가면 자동으로 계속 진행됩니다.' : '요청을 거부했습니다.', endpoint.endsWith('/approve') ? 'ok' : 'bad');
  } catch (error) {
    approveButton.disabled = false;
    denyButton.disabled = false;
    setStatus(`처리 실패: ${error instanceof Error ? error.message : String(error)}`, 'bad');
  }
}

loginButton.addEventListener('click', () => auth.login());
approveButton.addEventListener('click', () => decide('/connect/api/device/approve'));
denyButton.addEventListener('click', () => decide('/connect/api/device/deny'));

(async () => {
  const userCode = currentUserCode();
  codeEl.textContent = userCode || '코드 없음';
  if (!userCode) {
    setStatus('CLI가 표시한 승인 URL을 다시 열어 주세요.', 'bad');
    return;
  }
  try {
    const user = await auth.bootstrap();
    if (!user) {
      setStatus('NAKWOL ID로 로그인한 뒤 이 CLI 요청을 승인할 수 있습니다.');
      loginButton.hidden = false;
      return;
    }
    loginButton.hidden = true;
    approveButton.hidden = false;
    denyButton.hidden = false;
    setStatus(`${user.display_name || user.id} 계정으로 로그인됨. 승인하면 이 PC의 CLI에 30일짜리 Connect 접근 토큰이 발급됩니다.`);
  } catch (error) {
    setStatus(`로그인 확인 실패: ${error?.message || String(error)}`, 'bad');
    loginButton.hidden = false;
  }
})();
